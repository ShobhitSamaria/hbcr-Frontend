--------------------------------------------------------------------------------
1. DATABASE NAME
--------------------------------------------------------------------------------
hbcr_registry

Rationale: Project is the National HBCR Cancer Registry Portal. Lowercase +
underscore is the PostgreSQL convention. Use a single dedicated database so
backups, restores and connection pooling are simple.

--------------------------------------------------------------------------------
2. SCHEMA STRUCTURE
--------------------------------------------------------------------------------
Single schema: hbcr

  hbcr/
    reference tables  -> centres, hospitals, users
    patients          -> patients + side tables (ids, relatives, addresses,
                        habits, comorbidities)
    registrations     -> top-level HBCR registration record
    medical           -> pathological_diagnoses, diagnostic_methods,
                        diagnostic_procedures, treatments,
                        treatment_modality_details, familial_cancer_history
    enums             -> all dropdown values modelled as PG enums inside hbcr

If the project grows (auth, audit, file storage), promote those into separate
schemas (hbcr_auth, hbcr_audit) without touching the registry tables.

5. PRIMARY KEYS AND FOREIGN KEYS
--------------------------------------------------------------------------------

Primary keys:
  All tables: SERIAL `id`. The only natural candidate (hbcr_registration_no)
  also gets a UNIQUE constraint in addition.

Foreign keys (with ON DELETE action):

  hospitals.centre_id             -> centres.id                  ON DELETE RESTRICT
  users.hospital_id               -> hospitals.id                ON DELETE SET NULL
  patients.-- (no parent)                                           
  patient_identifications.patient_id -> patients.id              ON DELETE CASCADE
  patient_relatives.patient_id       -> patients.id              ON DELETE CASCADE
  patient_addresses.patient_id       -> patients.id              ON DELETE CASCADE
  patient_habits.patient_id          -> patients.id              ON DELETE CASCADE
  patient_comorbidities.patient_id   -> patients.id              ON DELETE CASCADE
  registrations.patient_id           -> patients.id              ON DELETE CASCADE
  registrations.hospital_id          -> hospitals.id             ON DELETE RESTRICT
  registrations.created_by_user_id   -> users.id                 ON DELETE SET NULL
  diagnostic_methods.registration_id   -> registrations.id       ON DELETE CASCADE
  diagnostic_procedures.diagnostic_method_id -> diagnostic_methods.id ON DELETE CASCADE
  pathological_diagnoses.registration_id    -> registrations.id       ON DELETE CASCADE
  treatments.registration_id                -> registrations.id       ON DELETE CASCADE
  treatment_modality_details.treatment_id   -> treatments.id          ON DELETE CASCADE
  familial_cancer_history.registration_id   -> registrations.id       ON DELETE CASCADE

8. NULLABILITY GUIDE
--------------------------------------------------------------------------------

NN (required):
  patients.full_name, patients.gender
  registrations.hbcr_registration_no, registrations.hospital_id, registrations.status
  diagnostic_methods.method
  diagnostic_procedures.procedure_name
  treatments.treatment_stage
  treatment_modality_details.modality
  centres.code, hospitals.name
  users.full_name, users.role, users.initials

NULL (optional, app may collect later):
  patients.age / date_of_birth  (one of them is usually required by UI)
  registration referral sub-fields (only if referral_type='other_hospital')
  anthropometric_* (UI marks required - enforce on app)
  diagnostic_procedures date (only ticked rows get a date)
  treatment modality date columns (UI spans for surgery)
  familial_cancer_history detail fields  (only meaningful when family_history='yes')

Indexed notes:
  - The patient row is created when "name" is provided; everything else can
    arrive in any order across side tables.

Key concepts:
- patients is the personal-data anchor; many 1:N child tables hang off it.
- registrations is the clinical-data anchor (1:N with patients; 1:N with
  diagnostic_methods, treatments; 1:1 with pathological_diagnoses and
  familial_cancer_history).
- treatments is also 1:N with treatment_modality_details (8 modalities x 2
  stages: prior_registration vs at_ri).
- hospitals connects to users, registrations, centres - multi-tenant.

-- 11.2 TABLES ------------------------------------------------------

-- centres
CREATE TABLE IF NOT EXISTS hbcr.centres (
  id          SERIAL PRIMARY KEY,
  code        VARCHAR(32)  NOT NULL UNIQUE,
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);

-- hospitals
CREATE TABLE IF NOT EXISTS hbcr.hospitals (
  id          SERIAL PRIMARY KEY,
  name        VARCHAR(255) NOT NULL,
  centre_id   INTEGER      NOT NULL REFERENCES hbcr.centres(id) ON DELETE RESTRICT,
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_hospitals_centre ON hbcr.hospitals(centre_id);

-- users
CREATE TABLE IF NOT EXISTS hbcr.users (
  id          SERIAL PRIMARY KEY,
  full_name   VARCHAR(255) NOT NULL,
  role        VARCHAR(64)  NOT NULL,
  initials    VARCHAR(4)   NOT NULL,
  hospital_id INTEGER      REFERENCES hbcr.hospitals(id) ON DELETE SET NULL,
  is_active   BOOLEAN      NOT NULL DEFAULT true,
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
);

-- patients
CREATE TABLE IF NOT EXISTS hbcr.patients (
  id            SERIAL PRIMARY KEY,
  full_name     VARCHAR(255) NOT NULL,
  age           SMALLINT,
  date_of_birth DATE,
  gender        hbcr.gender_enum NOT NULL,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS idx_patients_name       ON hbcr.patients(full_name);
CREATE INDEX IF NOT EXISTS idx_patients_name_trgm  ON hbcr.patients USING gin (full_name gin_trgm_ops);

-- patient_identifications
CREATE TABLE IF NOT EXISTS hbcr.patient_identifications (
  id          SERIAL PRIMARY KEY,
  patient_id  INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  id_type     hbcr.id_type_enum NOT NULL,
  number      VARCHAR(64) NOT NULL,
  UNIQUE (patient_id, id_type)
);
CREATE INDEX IF NOT EXISTS idx_patids_value ON hbcr.patient_identifications(number);

-- patient_relatives
CREATE TABLE IF NOT EXISTS hbcr.patient_relatives (
  id             SERIAL PRIMARY KEY,
  patient_id     INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  relationship   hbcr.relationship_enum NOT NULL,
  name           VARCHAR(255),
  mobile_number  VARCHAR(15),
  UNIQUE (patient_id, relationship)
);

-- patient_addresses
CREATE TABLE IF NOT EXISTS hbcr.patient_addresses (
  id            SERIAL PRIMARY KEY,
  patient_id    INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  address_type  hbcr.address_type_enum NOT NULL,
  flat_house_no VARCHAR(64),
  street_road   VARCHAR(255),
  city          VARCHAR(64),
  district      VARCHAR(64),
  state         VARCHAR(64),
  pin_code      VARCHAR(6),
  mobile_number VARCHAR(15),
  email         VARCHAR(255),
  UNIQUE (patient_id, address_type),
  CONSTRAINT pin_format     CHECK (pin_code IS NULL OR pin_code     ~ '^[1-9][0-9]{5}$'),
  CONSTRAINT mobile_format  CHECK (mobile_number IS NULL OR mobile_number ~ '^[6-9][0-9]{9}$'),
  CONSTRAINT email_format   CHECK (email IS NULL OR email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$')
);

-- patient_habits
CREATE TABLE IF NOT EXISTS hbcr.patient_habits (
  id              SERIAL PRIMARY KEY,
  patient_id      INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  habit           hbcr.habit_enum NOT NULL,
  answer          hbcr.yes_no_unknown_enum NOT NULL DEFAULT 'no',
  duration_months SMALLINT,
  UNIQUE (patient_id, habit),
  CONSTRAINT habit_duration CHECK (answer <> 'yes' OR duration_months IS NOT NULL)
);

-- patient_comorbidities
CREATE TABLE IF NOT EXISTS hbcr.patient_comorbidities (
  id              SERIAL PRIMARY KEY,
  patient_id      INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  comorbidity     hbcr.comorbidity_enum NOT NULL,
  answer          hbcr.yes_no_unknown_enum NOT NULL DEFAULT 'no',
  duration_months SMALLINT,
  UNIQUE (patient_id, comorbidity),
  CONSTRAINT comorb_duration CHECK (answer <> 'yes' OR duration_months IS NOT NULL)
);

-- registrations
CREATE TABLE IF NOT EXISTS hbcr.registrations (
  id                                SERIAL PRIMARY KEY,
  patient_id                        INTEGER NOT NULL REFERENCES hbcr.patients(id) ON DELETE CASCADE,
  hbcr_registration_no              VARCHAR(20) NOT NULL UNIQUE,
  hospital_id                       INTEGER NOT NULL REFERENCES hbcr.hospitals(id) ON DELETE RESTRICT,
  department_name                   VARCHAR(128),
  unit_number                       VARCHAR(32),
  hospital_registration_no          VARCHAR(64),
  date_of_reporting                 DATE,
  case_registered_through           hbcr.case_through_enum,
  referral_type                     hbcr.referral_type_enum,
  referral_facility_name            VARCHAR(255),
  referral_facility_city            VARCHAR(64),
  referral_facility_district        VARCHAR(64),
  referral_facility_hospital_lab_nh VARCHAR(255),
  referral_facility_reg_date        DATE,
  date_of_first_diagnosis           DATE,
  anthropometric_height_cm          NUMERIC(5,2),
  anthropometric_weight_kg          NUMERIC(5,2),
  marital_status                    hbcr.marital_status_enum,
  education                         hbcr.education_enum,
  status                            hbcr.registration_status_enum NOT NULL DEFAULT 'active',
  form_completed_by                 VARCHAR(255),
  form_completion_date              DATE,
  created_by_user_id                INTEGER REFERENCES hbcr.users(id) ON DELETE SET NULL,
  created_at                        TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT hbcr_reg_no_format CHECK (hbcr_registration_no ~ '^HBCR-[0-9]{4}-[0-9]{4,5}$'),
  CONSTRAINT referral_subfields CHECK (
    referral_type <> 'other_hospital'
    OR (referral_facility_name IS NOT NULL AND referral_facility_city IS NOT NULL)
  )
);
CREATE INDEX IF NOT EXISTS idx_registrations_status   ON hbcr.registrations(status);
CREATE INDEX IF NOT EXISTS idx_registrations_hospital ON hbcr.registrations(hospital_id);
CREATE INDEX IF NOT EXISTS idx_registrations_patient  ON hbcr.registrations(patient_id);

-- diagnostic_methods
CREATE TABLE IF NOT EXISTS hbcr.diagnostic_methods (
  id                  SERIAL PRIMARY KEY,
  registration_id     INTEGER NOT NULL REFERENCES hbcr.registrations(id) ON DELETE CASCADE,
  method              hbcr.diagnostic_method_enum NOT NULL,
  clinical_only_date  DATE,
  UNIQUE (registration_id, method)
);

-- diagnostic_procedures
CREATE TABLE IF NOT EXISTS hbcr.diagnostic_procedures (
  id                   SERIAL PRIMARY KEY,
  diagnostic_method_id INTEGER NOT NULL REFERENCES hbcr.diagnostic_methods(id) ON DELETE CASCADE,
  procedure_name       VARCHAR(128) NOT NULL,
  is_others            BOOLEAN NOT NULL DEFAULT false,
  others_specify       VARCHAR(255),
  procedure_date       DATE,
  UNIQUE (diagnostic_method_id, procedure_name)
);

-- pathological_diagnoses
CREATE TABLE IF NOT EXISTS hbcr.pathological_diagnoses (
  id                                SERIAL PRIMARY KEY,
  registration_id                   INTEGER NOT NULL UNIQUE REFERENCES hbcr.registrations(id) ON DELETE CASCADE,
  longest_symptom_duration_months   SMALLINT,
  anatomical_site                   VARCHAR(128),
  pathology_slide_no                VARCHAR(64),
  primary_tumor_site                VARCHAR(128),
  morphology                        VARCHAR(128),
  icdo_topography                   VARCHAR(64),
  icdo_morphology                   VARCHAR(64),
  secondary_site                    VARCHAR(128),
  metastasis_morphology             VARCHAR(128),
  icd10_site                        VARCHAR(64),
  laterality                        hbcr.laterality_enum DEFAULT 'not_paired_site',
  paired_laterality                 hbcr.paired_laterality_enum,
  sequence                          hbcr.sequence_enum
);
CREATE INDEX IF NOT EXISTS idx_pathdx_icdo_topo   ON hbcr.pathological_diagnoses(icdo_topography);
CREATE INDEX IF NOT EXISTS idx_pathdx_icdo_morph  ON hbcr.pathological_diagnoses(icdo_morphology);
CREATE INDEX IF NOT EXISTS idx_pathdx_icd10       ON hbcr.pathological_diagnoses(icd10_site);

-- treatments
CREATE TABLE IF NOT EXISTS hbcr.treatments (
  id                              SERIAL PRIMARY KEY,
  registration_id                 INTEGER NOT NULL REFERENCES hbcr.registrations(id) ON DELETE CASCADE,
  treatment_stage                 hbcr.treatment_stage_enum NOT NULL,
  treatment_given_choice          hbcr.yes_no_unknown_enum,
  treatment_type                  hbcr.treatment_type_enum,
  clinical_extent_of_disease      hbcr.clinical_extent_enum,
  staging_system                  hbcr.staging_system_enum,
  tnm_t                           VARCHAR(8),
  tnm_n                           VARCHAR(8),
  tnm_m                           VARCHAR(8),
  composite_stage                 VARCHAR(16),
  ecog_status                     hbcr.ecog_status_enum,
  ecog_grade                      hbcr.ecog_grade_enum,
  targeted_therapy_type           hbcr.targeted_therapy_enum,
  targeted_therapy_other_specify  VARCHAR(255),
  UNIQUE (registration_id, treatment_stage)
);

-- treatment_modality_details
CREATE TABLE IF NOT EXISTS hbcr.treatment_modality_details (
  id                 SERIAL PRIMARY KEY,
  treatment_id       INTEGER NOT NULL REFERENCES hbcr.treatments(id) ON DELETE CASCADE,
  modality           hbcr.treatment_modality_enum NOT NULL,
  is_selected        BOOLEAN NOT NULL DEFAULT false,
  intention_to_treat hbcr.intention_enum,
  role               hbcr.role_enum,
  details            hbcr.treatment_detail_enum,
  start_date         DATE,
  end_date           DATE,
  others_specify     VARCHAR(255),
  UNIQUE (treatment_id, modality)
);
CREATE INDEX IF NOT EXISTS idx_treatment_modality_sel
  ON hbcr.treatment_modality_details(treatment_id) WHERE is_selected = true;

-- familial_cancer_history
CREATE TABLE IF NOT EXISTS hbcr.familial_cancer_history (
  id                          SERIAL PRIMARY KEY,
  registration_id             INTEGER NOT NULL UNIQUE REFERENCES hbcr.registrations(id) ON DELETE CASCADE,
  family_history              hbcr.yes_no_unknown_enum NOT NULL DEFAULT 'no',
  relationship_with_cancer    hbcr.fam_relationship_enum,
  degree_of_relationship      hbcr.fam_degree_enum,
  primary_site                hbcr.fam_primary_site_enum,
  age_at_diagnosis            SMALLINT,
  date_of_diagnosis           DATE,
  CONSTRAINT fam_history_subfields CHECK (
    family_history <> 'yes'
    OR (relationship_with_cancer IS NOT NULL AND degree_of_relationship IS NOT NULL)
  )
);
